3. Comparision Operators :: Data Comparision :: boolean :: true/false

1. EqualsTo :: == :: To check equality :: only values
2. Not EqualsTo :: != :: to check inequality  :: only values
3. LessThan :: < :: Left value is less than right value
4. LessThanOrEqual :: <= :: Left value is less than or equal to right value
5. GreaterThan :: > :: Left value is GreaterThan the right value
6. GreaterThanOrEqual :: >= :: Left value is GreaterThan or equal to right value
7. StrictlyEqualTo :: === :: To check equality :: value and type
8. Strictly Not EqualTo :: !== :: To check inequality :: value and type


5. Ternary Operators :: Conditioanl Operator :: ? :: 

Syntax ::

Example ::

4. Logical Operators :: boolean :: true/false

1. Logical AND Operator :: && :: 
2. Logical OR Operator :: || :: 
3. Logical NOT Operator :: ! :: 

1. Logical AND Operator :: && :: 
        if all inputs true returns true, else false.

ip1       &&            ip2             op
true                    true            true
true                    false           false
false                   true            false
false                   false           false

2. Logical OR Operator :: || :: 
        if atleast one input is true returns true, else false.

ip1       ||            ip2             op
true                    true            true
true                    false           true
false                   true            true
false                   false           false

3. Logical NOT Operator :: ! ::   opposite to input

        !true == false
        !false == true
